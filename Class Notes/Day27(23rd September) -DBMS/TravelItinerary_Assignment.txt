3. Places
Column		Data Type	Constraint		Description
place_id	INT		PK, AUTO_INCREMENT	Place ID
destination_id	INT		FK			Destination
place_name	VARCHAR(100)	NOT NULL		Tourist attraction
place_type	VARCHAR(50)	—		Beach, Historical, Nature etc.
entry_fee	DECIMAL(10,2)	DEFAULT 0	Entry fee
rating		DECIMAL(3,2)	—		Rating

4. activities
Stores activities that can be performed at tourist places.
Column		Data Type	Constraint		Description
activity_id	INT		PK, AUTO_INCREMENT	Activity ID
place_id	INT		FK		Place where activity occurs
activity_name	VARCHAR(100)	NOT NULL		Activity name
duration_hours	DECIMAL(4,2)	—			Duration
cost		DECIMAL(10,2)	—			Activity cost
difficulty	VARCHAR(30)	—			Easy/Medium/Hard

CREATE TABLE activities (
    activity_id INT PRIMARY KEY AUTO_INCREMENT,
    place_id INT NOT NULL,
    activity_name VARCHAR(100) NOT NULL,      
    duration_minutes INT,
    cost DECIMAL(8, 2) DEFAULT 0.00,           
    description TEXT,
    foreign key place_id references places(place_id) 
);

5. trips
Stores trips created by users.
Column		Data Type	Constraint		Description
trip_id		INT		PK, AUTO_INCREMENT	Trip ID
user_id		INT		FK			Trip owner
destination_id	INT		FK			Destination
trip_name	VARCHAR(150)	NOT NULL		Trip name
start_date	DATE		NOT NULL		Starting date
end_date	DATE		NOT NULL		Ending date
budget		DECIMAL(12,2)	—			Planned budget
status		VARCHAR(30)	DEFAULT			Trip status
